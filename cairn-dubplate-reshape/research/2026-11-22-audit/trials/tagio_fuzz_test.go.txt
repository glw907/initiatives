package tagio

import (
	"os"
	"path/filepath"
	"testing"
)

// FuzzParseBounded feeds arbitrary bytes to the bounded container walk
// and to Read. Either may refuse; neither may panic, and a refusal must
// carry the zero Container.
func FuzzParseBounded(f *testing.F) {
	golden, err := os.ReadFile(filepath.Join("testdata", "golden.flac"))
	if err != nil {
		f.Fatal(err)
	}
	f.Add(golden)
	f.Add(golden[:64])
	f.Add([]byte("fLaC"))
	f.Add([]byte{})
	f.Fuzz(func(t *testing.T, data []byte) {
		p := filepath.Join(t.TempDir(), "in.flac")
		if err := os.WriteFile(p, data, 0o600); err != nil {
			t.Fatal(err)
		}
		c, err := ParseBounded(p, Bounds{TagBytes: 1 << 16, PictureBytes: 1 << 18})
		if err != nil && (len(c.Blocks) != 0 || c.FrameLength != 0) {
			t.Fatalf("refusal returned a non-zero Container: %v", err)
		}
		if err == nil && c.FrameOffset+c.FrameLength > int64(len(data)) {
			t.Fatalf("frame region %d+%d runs past %d bytes", c.FrameOffset, c.FrameLength, len(data))
		}
		_, _ = Read(p)
		_, _ = Sniff(p)
	})
}
